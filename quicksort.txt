#include <iostream>
#include <vector>
#include <algorithm>
#include <cassert>

void quick_sort(std::vector<int>& arr, int left, int right) {
    if (left >= right) return;
    int pivot = arr[(left + right) / 2];
    int i = left, j = right;
    while (i <= j) {
        while (arr[i] < pivot) i++;
        while (arr[j] > pivot) j--;
        if (i <= j) {
            std::swap(arr[i], arr[j]);
            i++;
            j--;
        }
    }
    if (left < j) quick_sort(arr, left, j);
    if (i < right) quick_sort(arr, i, right);
}

void quick_sort(std::vector<int>& arr) {
    if (!arr.empty())
        quick_sort(arr, 0, arr.size() - 1);
}

int main() {
    // Test 1: general unsorted array
    {
        std::vector<int> arr = {3, 1, 4, 1, 5, 9, 2, 6, 5};
        std::vector<int> expected = arr;
        std::sort(expected.begin(), expected.end());
        quick_sort(arr);
        assert(arr == expected);
    }

    // Test 2: empty array
    {
        std::vector<int> arr;
        quick_sort(arr);
        assert(arr.empty());
    }

    // Test 3: single element
    {
        std::vector<int> arr = {10};
        quick_sort(arr);
        assert(arr[0] == 10);
    }

    // Test 4: array with negative values
    {
        std::vector<int> arr = {5, -1, 3, 0, 2, -2, 1, 4};
        std::vector<int> expected = arr;
        std::sort(expected.begin(), expected.end());
        quick_sort(arr);
        assert(arr == expected);
    }

    std::cout << "All tests passed!" << std::endl;
    return 0;
}
